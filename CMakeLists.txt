cmake_minimum_required(VERSION 3.16)
project(multi_layer_ndt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(WITH_UI "Build the copied Pangolin viewer" ON)
option(BUILD_TESTING "Build registration and mapping regression tests" OFF)
if(BUILD_TESTING)
  enable_testing()
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosbag2_storage REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(GTSAM REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree search)
find_package(OpenMP REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(glog REQUIRED)
find_package(Threads REQUIRED)
if(WITH_UI)
  find_package(Pangolin REQUIRED)
  find_package(OpenGL REQUIRED)
endif()
add_subdirectory(src)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
